function calculateNextInterval(card, quality) {
    if (quality < 2) {
        // Если ответ плохой - сброс
        card.interval = 1; // Повторить завтра
        card.streak = 0;
        card.easeFactor = Math.max(1.3, card.easeFactor - 0.2);
    } else {
        // Если ответ хороший
        card.streak++;
        
        if (card.streak == 1) {
            card.interval = 1;
        } else if (card.streak == 2) {
            card.interval = 3;
        } else {
            // Основная формула SM-2 (модифицированная)
            card.interval = Math.round(card.interval * card.easeFactor);
        }
        
        // Корректировка easeFactor на основе оценки
        if (quality == 2) {
            card.easeFactor = Math.max(1.3, card.easeFactor - 0.15);
        } else if (quality == 3) {
            card.easeFactor = card.easeFactor; // Без изменений
        } else if (quality == 4) {
            card.easeFactor = card.easeFactor + 0.1;
        }
    }
    
    // Ограничиваем интервал разумными пределами (макс 180 дней)
    card.interval = Math.min(card.interval, 180);
    
    // Устанавливаем дату следующего повторения
    card.nextReviewDate = currentDate + card.interval;
    
    return card;
}

function getOptimalOrderForCategory(categoryCards) {
    // 1. Сначала проблемные карточки (низкий easeFactor)
    const difficult = categoryCards.filter(c => c.easeFactor < 1.8)
                                   .sort((a, b) => a.easeFactor - b.easeFactor);
    
    // 2. Потом карточки, связанные концептуально
    const conceptualGroups = groupConceptually(categoryCards);
    
    // 3. Чередуем подкатегории для лучшего запоминания
    const ordered = [];
    const subcats = getUniqueSubcategories(categoryCards);
    
    // Round-robin по подкатегориям
    let index = 0;
    while (ordered.length < categoryCards.length) {
        const subcat = subcats[index % subcats.length];
        const cardFromSubcat = categoryCards.find(c => 
            c.subcategory === subcat && !ordered.includes(c)
        );
        
        if (cardFromSubcat) {
            ordered.push(cardFromSubcat);
        }
        index++;
    }
    
    return [...difficult, ...ordered];
}

    // 2. Карточки с похожими ключевыми словами
    const keywordMatches = findCardsByKeywords(card.question, allCards);
    
    // 3. Карточки, которые часто путают (на основе истории)
    const commonlyConfused = getCommonlyConfused(card.id, allCards);
    
    // Комбинируем с приоритетами
    return [
        ...sameSubcategory.slice(0, 2),
        ...keywordMatches.slice(0, 1),
        ...commonlyConfused.slice(0, 1)
    ].slice(0, count);
}

class CategoryProgress {
    constructor() {
        this.categoryStats = {};
    }
    
    updateAfterSession(sessionCards) {
        sessionCards.forEach(card => {
            const cat = card.category;
            const subcat = card.subcategory;
            
            if (!this.categoryStats[cat]) {
                this.categoryStats[cat] = {
                    totalCards: 0,
                    learnedCards: 0,
                    averageEase: 2.5,
                    subcategories: {}
                };
            }
            
            if (!this.categoryStats[cat].subcategories[subcat]) {
                this.categoryStats[cat].subcategories[subcat] = {
                    learned: 0,
                    total: countCardsInSubcategory(subcat),
                    nextReviewDate: null
                };
            }
            
            // Обновляем статистику
            if (card.repetitionCount > 0) {
                this.categoryStats[cat].learnedCards++;
                this.categoryStats[cat].subcategories[subcat].learned++;
            }
        });
    }
    
    getWeakestCategory() {
        // Возвращает категорию с наименьшим процентом освоения
        return Object.entries(this.categoryStats)
            .map(([cat, stats]) => ({
                category: cat,
                mastery: stats.learnedCards / stats.totalCards,
                weakestSubcat: this.getWeakestSubcategory(cat)
            }))
            .sort((a, b) => a.mastery - b.mastery)[0];
    }
}

function generateDailyPlan(cards, dayNumber) {
    const plan = {
        focusCategory: null,
        focusSubcategory: null,
        cards: [],
        sessionType: 'mixed' // 'theme', 'mixed', 'review'
    };
    
    // Определяем тип сессии на основе дня
    if (dayNumber % 7 === 0) {
        // Воскресенье - обзор слабых мест
        plan.sessionType = 'review';
        plan.focusCategory = getWeakestCategory().category;
        plan.cards = getWeakCardsFromCategory(plan.focusCategory, 20);
    } else if (dayNumber % 3 === 0) {
        // Каждые 3 дня - тематическая сессия
        plan.sessionType = 'theme';
        plan.focusSubcategory = getNextSubcategoryToMaster();
        plan.cards = getAllCardsFromSubcategory(plan.focusSubcategory);
    } else {
        // Обычные дни - смешанная сессия
        plan.sessionType = 'mixed';
        plan.cards = getMixedSession(cards, {
            minCategories: 4,
            cardsPerCategory: { min: 3, max: 8 }
        });
    }
    
    // Добавляем "карточки-мостики" между темами
    if (plan.sessionType === 'mixed') {
        plan.cards = addBridgeCards(plan.cards, cards);
    }
    
    return plan;
}

12. Преимущества этого подхода
Контекстное обучение - связанные концепции изучаются вместе

Прочное структурирование знаний - информация организуется в ментальные модели

Выявление пробелов - видно, какие темы вызывают трудности

Эффективные сравнения - можно сравнивать схожие концепции из разных категорий

Адаптивность - система фокусируется на слабых местах в структуре знаний

Этот алгоритм превращает простые карточки в целостную систему обучения, где каждая карточка не изолирована, а является частью смысловой сети. Это ускоряет обучение и улучшает понимание связей между концепциями.
